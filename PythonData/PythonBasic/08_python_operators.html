Python Basics · Lesson 08

연산자와 연산 순서

연산자는 Python 객체를 계산하고 비교하며, 여러 조건을 결합하고 비트를 조작한 뒤 결과를 다시 변수 이름에 연결하는 문법입니다. 이번 단원에서는 산술 연산자인 +, -, *, /, //, %, **와 비교·논리·비트·시프트· 포함·동일성·할당 연산자를 공부합니다. 또한 괄호와 연산자 우선순위, 거듭제곱의 오른쪽 결합, 비교 체이닝, 논리 연산의 단축 평가, 가변 객체의 복합 할당을 함께 정리합니다.

Operator Operand Expression Arithmetic Comparison Logical Bitwise Shift Assignment Short-circuit Precedence Associativity

이 단원의 중심 질문

여러 연산자가 하나의 표현식에 섞이면 Python은 무엇부터 계산할까요?

Python은 단순히 왼쪽부터 모든 연산을 실행하지 않습니다. 괄호와 연산자 우선순위에 따라 표현식을 먼저 묶고, 그 구조에 따라 값을 평가합니다. 또한 같은 연산자라도 숫자, 문자열, 리스트, 집합, NumPy 배열처럼 피연산자의 자료형에 따라 의미가 달라질 수 있습니다.

우선순위와 평가 순서는 다르다

우선순위는 표현식을 어떻게 묶을지 결정하고, 평가 순서는 각 피연산자를 언제 실행할지 결정합니다.

    print(label)
    return value

    show("왼쪽", 10)
    + show("오른쪽", 20)

오른쪽

연산자와 연산 순서 종합 예제

다음 코드는 산술, 음수의 몫과 나머지, 비교 체이닝, 논리 연산, 비트 권한, 시프트, 복합 할당과 우선순위를 하나의 실행 파일에서 확인합니다.

파일명: operators_overview.py 

from __future__ import annotations


PASS_SCORE = 80
FREE_DELIVERY_AMOUNT = 30000

READ = 1 << 0
WRITE = 1 << 1
DELETE = 1 << 2


def print_section(
    title: str,
) -> None:
    """실습 영역 제목을 출력한다."""

    print("\n" + "=" * 60)
    print(title)
    print("=" * 60)


def demonstrate_arithmetic() -> None:
    """산술 연산자를 확인한다."""

    print_section("1. 산술 연산")

    left = 10
    right = 3

    print("덧셈:", left + right)
    print("뺄셈:", left - right)
    print("곱셈:", left * right)
    print("나눗셈:", left / right)
    print("바닥 나눗셈:", left // right)
    print("나머지:", left % right)
    print("거듭제곱:", left ** right)


def demonstrate_floor_modulo() -> None:
    """음수의 몫과 나머지를 확인한다."""

    print_section("2. 음수의 //와 %")

    cases = [
        (-7, 3),
        (7, -3),
        (-7, -3),
    ]

    for dividend, divisor in cases:
        quotient = (
            dividend
            // divisor
        )

        remainder = (
            dividend
            % divisor
        )

        restored = (
            quotient * divisor
            + remainder
        )

        print(
            f"{dividend} // {divisor}"
            f" = {quotient}"
        )

        print(
            f"{dividend} % {divisor}"
            f" = {remainder}"
        )

        print("복원:", restored)


def demonstrate_comparison() -> None:
    """비교와 포함 연산을 확인한다."""

    print_section("3. 비교와 포함")

    score = 85

    print(
        "통과 여부:",
        score >= PASS_SCORE,
    )

    print(
        "유효 범위:",
        0 <= score <= 100,
    )

    skills = {
        "Python",
        "SQL",
        "Git",
    }

    print(
        "Python 포함:",
        "Python" in skills,
    )


def demonstrate_logical() -> None:
    """논리 연산과 피연산자 반환을 확인한다."""

    print_section("4. 논리 연산")

    is_logged_in = True
    has_permission = True
    is_banned = False

    can_access = (
        is_logged_in
        and has_permission
        and not is_banned
    )

    print("접근 가능:", can_access)

    input_name = ""

    display_name = (
        input_name
        or "익명 사용자"
    )

    print("표시 이름:", display_name)


def demonstrate_bitwise() -> None:
    """비트 연산과 권한을 확인한다."""

    print_section("5. 비트 연산")

    print("4 & 5:", 4 & 5)
    print("4 | 5:", 4 | 5)
    print("4 ^ 5:", 4 ^ 5)
    print("~4:", ~4)

    permissions = READ | WRITE

    has_read = (
        permissions & READ
    ) != 0

    has_write = (
        permissions & WRITE
    ) != 0

    has_delete = (
        permissions & DELETE
    ) != 0

    print("읽기:", has_read)
    print("쓰기:", has_write)
    print("삭제:", has_delete)


def demonstrate_shift() -> None:
    """시프트 연산을 확인한다."""

    print_section("6. 시프트 연산")

    number = 5
    count = 2

    print(
        "왼쪽 시프트:",
        number << count,
    )

    print(
        "곱셈 비교:",
        number * (2 ** count),
    )

    print(
        "오른쪽 시프트:",
        number >> count,
    )

    print(
        "나눗셈 비교:",
        number // (2 ** count),
    )


def demonstrate_assignment() -> None:
    """복합 할당과 객체 공유를 확인한다."""

    print_section("7. 복합 할당")

    count = 10
    count += 1

    print("count:", count)

    list_a = [1, 2]
    list_b = list_a

    list_a += [3]

    print("list_a:", list_a)
    print("list_b:", list_b)
    print(
        "같은 객체:",
        list_a is list_b,
    )


def demonstrate_precedence() -> None:
    """연산자 우선순위와 결합 방향을 확인한다."""

    print_section("8. 연산자 우선순위")

    print(
        "10 + 2 * 3:",
        10 + 2 * 3,
    )

    print(
        "(10 + 2) * 3:",
        (10 + 2) * 3,
    )

    print(
        "2 ** 3 ** 2:",
        2 ** 3 ** 2,
    )

    print(
        "(2 ** 3) ** 2:",
        (2 ** 3) ** 2,
    )

    print(
        "-2 ** 2:",
        -2 ** 2,
    )

    print(
        "(-2) ** 2:",
        (-2) ** 2,
    )


def demonstrate_order() -> None:
    """주문 업무에 연산자를 적용한다."""

    print_section("9. 주문 계산")

    unit_price = 12000
    quantity = 3
    discount = 2000

    subtotal = (
        unit_price
        * quantity
    )

    final_price = (
        subtotal
        - discount
    )

    is_free_delivery = (
        final_price
        >= FREE_DELIVERY_AMOUNT
    )

    can_order = (
        quantity > 0
        and final_price > 0
    )

    print("상품 금액:", subtotal)
    print("최종 금액:", final_price)
    print("무료 배송:", is_free_delivery)
    print("주문 가능:", can_order)


def main() -> None:
    """연산자 실습을 순서대로 실행한다."""

    demonstrate_arithmetic()
    demonstrate_floor_modulo()
    demonstrate_comparison()
    demonstrate_logical()
    demonstrate_bitwise()
    demonstrate_shift()
    demonstrate_assignment()
    demonstrate_precedence()
    demonstrate_order()


if __name__ == "__main__":
    main()

14 · Common Mistakes

초보자가 자주 하는 연산자 실수

01

모든 연산을 왼쪽부터 계산합니다

곱셈·나눗셈은 덧셈·뺄셈보다 먼저 계산됩니다.

02

=과 ==를 혼동합니다

=은 할당, ==는 값 비교입니다.

03

/가 항상 정수를 반환한다고 생각합니다

Python 3의 일반 나눗셈은 보통 float을 반환합니다.

04

//를 단순 소수점 제거로 이해합니다

음의 무한대 방향으로 내리는 바닥 나눗셈입니다.

05

음수의 나머지를 직관으로만 판단합니다

몫과 나머지의 복원 관계를 이용해 결과를 확인해야 합니다.

06

거듭제곱을 왼쪽부터 결합합니다

**는 오른쪽부터 결합합니다.

07

음수 제곱에 괄호를 생략합니다

-2 ** 2와 (-2) ** 2는 다릅니다.

08

and와 &를 같은 연산으로 봅니다

논리 조건과 비트·마스크 연산은 의미와 평가 방식이 다릅니다.

09

and와 or가 항상 bool을 반환한다고 생각합니다

논리 평가에 사용된 피연산자를 그대로 반환할 수 있습니다.

10

단축 평가를 고려하지 않습니다

오른쪽 함수나 메서드가 실행되지 않을 수 있습니다.

11

딕셔너리의 in이 값을 찾는다고 생각합니다

기본적으로 딕셔너리 키를 검사합니다.

12

비교 체이닝을 왼쪽 결합으로 이해합니다

연속 비교는 Python의 특별한 체이닝 규칙을 사용합니다.

13

리스트의 +=가 항상 새 객체를 만든다고 생각합니다

기존 리스트 객체를 직접 확장할 수 있습니다.

17 · Interview

면접 포인트

Python 연산자란 무엇입니까?

객체를 계산·비교·결합하거나 결과를 이름에 다시 할당하기 위한 기호 또는 키워드입니다. 연산 대상은 피연산자이며, 평가되어 결과를 만드는 코드는 표현식입니다.

단항 연산자와 이항 연산자의 차이는 무엇입니까?

단항 연산자는 피연산자 하나를 사용하고, 이항 연산자는 두 개를 사용합니다. -value, not value, ~value는 단항 연산이며, a + b, a > b, a and b는 이항 연산입니다.

/와 //의 차이는 무엇입니까?

/는 일반 나눗셈으로 일반적으로 float을 반환합니다. //는 바닥 나눗셈으로 결과를 음의 무한대 방향으로 내립니다.

왜 -7 // 3은 -3입니까?

//는 소수 부분을 0 방향으로 제거하는 것이 아니라 바닥값을 구하기 때문입니다. -2.333...보다 작거나 같은 가장 가까운 정수는 -3입니다.

Python의 몫과 나머지는 어떤 관계를 유지합니까?

a == (a // b) * b + a % b 관계를 유지합니다. 음수가 포함되어도 이 공식으로 원래 값을 복원할 수 있습니다.

divmod()는 무엇을 반환합니까?

바닥 나눗셈의 몫과 나머지를 (a // b, a % b) 형태의 튜플로 반환합니다.

2 ** 3 ** 2의 결과는 무엇입니까?

거듭제곱은 오른쪽부터 결합하므로 2 ** (3 ** 2), 즉 2 ** 9가 되어 결과는 512입니다.

-2 ** 2와 (-2) ** 2의 차이는 무엇입니까?

-2 ** 2는 -(2 ** 2)로 평가되어 -4이고, (-2) ** 2는 음수 전체를 제곱하여 4입니다.

=과 ==의 차이는 무엇입니까?

=은 이름과 객체를 연결하는 할당 연산자이고, ==는 두 객체의 값이나 내용이 같은지 비교합니다.

비교 체이닝이란 무엇입니까?

0 <= score <= 100처럼 비교 연산을 연결하는 문법입니다. 두 조건을 논리적으로 함께 검사하며, 중간 표현식을 한 번만 평가합니다.

==와 is의 차이는 무엇입니까?

==는 값이나 내용의 동등성을, is는 두 이름이 정확히 같은 객체를 가리키는지 확인합니다.

딕셔너리에서 in은 무엇을 검사합니까?

기본적으로 키의 존재 여부를 검사합니다. 값은 value in dictionary.values() 형태로 확인합니다.

논리 연산자의 우선순위는 무엇입니까?

not이 가장 높고, 다음이 and, 마지막이 or입니다.

and와 or는 항상 bool을 반환합니까?

아닙니다. 논리 평가 결과에 따라 피연산자 중 하나를 그대로 반환합니다. not은 항상 bool을 반환합니다.

단축 평가란 무엇입니까?

논리 연산 결과가 이미 결정되면 뒤쪽 표현식을 실행하지 않는 방식입니다. 거짓인 값 뒤의 and 오른쪽과, 참인 값 뒤의 or 오른쪽은 평가하지 않습니다.

and와 &의 차이는 무엇입니까?

and는 일반 논리 조건을 결합하고 단축 평가를 수행합니다. &는 정수 비트, 집합 교집합 또는 NumPy·Pandas의 요소별 마스크 연산에 사용합니다.

NumPy 배열 조건에서 and를 사용할 수 없는 이유는 무엇입니까?

배열 비교 결과에는 여러 개의 True와 False가 있으므로 전체를 하나의 논리값으로 판단할 수 없기 때문입니다. 각 비교식을 괄호로 묶고 &, |, ~를 사용합니다.

왼쪽 시프트와 오른쪽 시프트는 수학적으로 어떻게 이해할 수 있습니까?

정수 x에 대해 x << n은 x * 2 ** n, x >> n은 x // 2 ** n과 연결해 이해할 수 있습니다.

a += b와 a = a + b는 항상 같습니까?

값의 결과는 같을 수 있지만 객체 처리 방식은 다를 수 있습니다. 리스트의 +=는 기존 리스트를 직접 변경할 수 있지만, a = a + b는 새 리스트를 만든 뒤 재할당합니다.

복잡한 표현식에서 괄호를 사용하는 이유는 무엇입니까?

연산 우선순위를 변경할 수 있을 뿐 아니라 개발자가 의도한 업무 규칙과 계산 순서를 읽는 사람에게 명확하게 보여 주기 위해서입니다.
